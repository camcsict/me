const ADMIN_SECRET_KEY = "LINGO_ADMIN_2026";

function doGet(e) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const action = (e && e.parameter && e.parameter.action) ? e.parameter.action : "lessons";
  const clientKey = (e && e.parameter && e.parameter.adminKey) ? e.parameter.adminKey : "";

  if (action === "admin_logs") {
    if (clientKey !== ADMIN_SECRET_KEY) {
      return ContentService.createTextOutput(JSON.stringify({ status: "forbidden" }))
        .setMimeType(ContentService.MimeType.JSON);
    }
    const logSheet = ss.getSheetByName("LichSuDiem");
    const logRows = logSheet ? logSheet.getDataRange().getValues() : [];
    return ContentService.createTextOutput(JSON.stringify({ status: "ok", logs: logRows.slice(1).reverse().slice(0, 50) }))
      .setMimeType(ContentService.MimeType.JSON);
  }

  const sheet = ss.getSheetByName("DanhSachBaiHoc") || ss.getSheets()[0];
  const rows = sheet.getDataRange().getValues();
  const headers = rows[0].map(h => String(h).trim());
  const lessons = [];
  for (let i = 1; i < rows.length; i++) {
    const item = {};
    headers.forEach((k, idx) => { if (k) item[k] = rows[i][idx]; });
    const isActive = String(item.status || "TRUE").toUpperCase() !== "FALSE";
    if ((isActive || clientKey === ADMIN_SECRET_KEY) && item.youtube_input) {
      lessons.push(item);
    }
  }
  return ContentService.createTextOutput(JSON.stringify({ status: "ok", lessons }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const d = JSON.parse(e.postData.contents);

  if (d.action === "add_lesson") {
    if (d.adminKey !== ADMIN_SECRET_KEY) {
      return ContentService.createTextOutput(JSON.stringify({ status: "forbidden" }))
        .setMimeType(ContentService.MimeType.JSON);
    }
    const sheet = ss.getSheetByName("DanhSachBaiHoc") || ss.getSheets()[0];
    sheet.appendRow([
      d.id, d.lang, d.title_vi, d.title_en, d.youtube_input, "",
      d.req_sec, d.xp, d.quiz_vi, d.quiz_en || d.quiz_vi, d.options, 0, "TRUE", "Mới", ""
    ]);
    return ContentService.createTextOutput(JSON.stringify({ status: "lesson_added" }))
      .setMimeType(ContentService.MimeType.JSON);
  }

  let logSheet = ss.getSheetByName("LichSuDiem") || ss.insertSheet("LichSuDiem");
  logSheet.appendRow([new Date(), d.email, d.lessonId, d.watchedSec, d.xpEarned, d.lessonStreak]);
  return ContentService.createTextOutput(JSON.stringify({ status: "score_saved" }))
    .setMimeType(ContentService.MimeType.JSON);
}
